function generateNumber(regionCode){
    const letters = 'АВЕКМНОРСТУХ';
    const l1 = letters[Math.floor(Math.random()*letters.length)];
    const l2 = letters[Math.floor(Math.random()*letters.length)];
    const digits = Array.from({length:3},()=>Math.floor(Math.random()*10));
    let rarity = 'common';
    const allSame = digits.every(d=>d===digits[0]);
    const isMirror = digits[0]===digits[2];
    const isSequence = Math.abs(digits[0]-digits[1])===1 && Math.abs(digits[1]-digits[2])===1;
    const isRound = digits[1]===0 && digits[2]===0;
    if(digits[0]===0 && digits[1]===0 && digits[2]===7) rarity='secret';
    else if(allSame && digits[0]>=7) rarity='legendary';
    else if(allSame) rarity='epic';
    else if(isMirror && digits[0]!==0) rarity='rare';
    else if(isRound || (isSequence && digits[0]!==0)) rarity='uncommon';
    const number = `${l1} ${digits.join('')} ${l2}${l2} ${regionCode}`;
    return { number, rarity, regionCode };
}

function calculatePrice(n){
    const region = REGIONS.find(r=>r.code===n.regionCode);
    const base = RARITIES[n.rarity].basePrice;
    const mult = region?region.multiplier:1;
    return Math.floor(base*mult*(0.9+Math.random()*0.2));
}

                if(r.ok){ tg.showAlert('✅ Куплено!'); await loadPlayer(); updateUI(); loadAuction(); }
                else{ tg.showAlert(r.message); }
            };
        });
    }catch(e){ console.error(e); }
}

async function loadTop(){
    try{
        const list = await fetch(API+'/api/top').then(r=>r.json());
        const medals = ['🥇','🥈','🥉'];
        document.getElementById('topList').innerHTML = list.map((p,i)=>`
            <div class="rank-item">
                <div>${medals[i]||(i+1)+'.'} ${p.name}</div>
                <div style="color:#f0c040">${p.balance.toLocaleString('ru-RU')} ₽</div>
            </div>
        `).join('');
    }catch(e){ console.error(e); }
}

document.querySelectorAll('.tab').forEach(t=>{
    t.onclick = ()=>{
        document.querySelectorAll('.tab').forEach(x=>x.classList.remove('active'));
        document.querySelectorAll('.section').forEach(x=>x.classList.remove('active'));
        t.classList.add('active');
        document.getElementById('tab-'+t.dataset.tab).classList.add('active');
        if(t.dataset.tab==='auc') loadAuction();
        if(t.dataset.tab==='top') loadTop();
        if(t.dataset.tab==='cases') renderCases();
    };
});
